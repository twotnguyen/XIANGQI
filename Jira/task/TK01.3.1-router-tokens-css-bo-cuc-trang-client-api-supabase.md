// chưa có phiên Supabase ⇒ <Navigate to={`/login?next=${encodeURIComponent(location.pathname + location.search)}`} replace />

### 5.6 Client API — `apps/web/src/lib/api.ts`
```ts
export class ApiError extends Error { constructor(public code: string, message: string, public status: number) { super(message); } }
export async function apiFetch<T>(path: string, init: RequestInit = {}): Promise<T> {
  const token = (await supabase.auth.getSession()).data.session?.access_token;
  const res = await fetch(`${import.meta.env.VITE_API_URL}${path}`, {
    ...init, credentials: 'omit',
    headers: { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}), ...init.headers },
  });
  const body = await res.json().catch(() => null);
  if (body?.ok === true) return body.data as T;                       // vỏ ApiResult thành công
  const err = body?.error ?? { code: 'NETWORK_ERROR', message: 'Không kết nối được máy chủ' };
  throw new ApiError(err.code, err.message, res.status);
}
```
Máy chủ trả mọi kết quả trong vỏ [`ApiResult`](../05-TU-DIEN-KY-THUAT.md#api-result) `{ ok, data | error }` (TK03.1.2). `credentials: 'omit'` ⇒ không dùng cookie liên miền.
